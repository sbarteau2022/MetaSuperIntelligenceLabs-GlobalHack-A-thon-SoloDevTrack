```python
"""
governance.py -- shared governance schema for AIE-304-M05.

Two artifacts, one discipline: every field in both schemas below exists
because a real reviewer -- an auditor, a regulator, a security lead who
did not build this system -- would ask for it, and every claim field is
checked, not trusted, the same way AIE-304-M04's validate_finding()
checked a red-team finding instead of trusting its prose.

  - The risk register maps measured evidence (an AIE-304-M04 finding, or a
    raw AIE-304-M01/M02 lab measurement) onto NIST AI RMF 1.0's four
    functions -- GOVERN, MAP, MEASURE, MANAGE -- using a working subset of
    the framework's real subcategory IDs (e.g. "MEASURE 2.7"). Descriptions
    below are paraphrased for length; the IDs are the framework's actual
    numbering, not this module's invention.

  - The model card is scoped by the EU AI Act's (Regulation (EU)
    2024/1689) own four risk tiers -- unacceptable, high-risk, limited,
    minimal -- using the Act's real Annex III high-risk domains, Article 5
    prohibited-practice categories, and Annex IV technical-documentation
    point numbers 1-8.

Nothing here makes a network call, and nothing here is a substitute for
reading the actual regulation text or the actual RMF Playbook before
filing something with real legal weight.
"""
from __future__ import annotations

import re

# ---------------------------------------------------------------------------
# Carried over unmodified from AIE-304-M04 Lesson 3: severity is computed
# from two measured axes, never asserted. This module's risk register
# reuses the identical function -- a risk register that recomputed severity
# with a different matrix than the one that filed the finding would
# silently diverge from the evidence it claims to summarize.
# ---------------------------------------------------------------------------
IMPACT_LEVELS = ("informational", "low", "medium", "high", "critical")
LIKELIHOOD_LEVELS = ("none_observed", "low", "medium", "high", "certain")

_MATRIX = {
    "critical":      {"none_observed": "high",          "low": "critical", "medium": "critical",
                       "high": "critical",               "certain": "critical"},
    "high":          {"none_observed": "medium",         "low": "high",     "medium": "high",
                       "high": "critical",                "certain": "critical"},
    "medium":        {"none_observed": "low",            "low": "medium",   "medium": "medium",
                       "high": "high",                    "certain": "high"},
    "low":           {"none_observed": "informational",  "low": "low",      "medium": "low",
                       "high": "medium",                  "certain": "medium"},
    "informational": {lvl: "informational" for lvl in LIKELIHOOD_LEVELS},
}


def rate_severity(impact: str, likelihood: str) -> str:
    if impact not in _MATRIX:
        raise ValueError(f"unknown impact level: {impact!r}")
    if likelihood not in _MATRIX[impact]:
        raise ValueError(f"unknown likelihood level: {likelihood!r}")
    return _MATRIX[impact][likelihood]

# ---------------------------------------------------------------------------
# Risk register schema. source_evidence is the load-bearing field: it is
# not just required to be non-empty, it is required to be TRACEABLE --
# validate_risk_entry() runs check_evidence_traceable() against it, so a
# fluent paragraph that cites nothing checkable fails exactly the way an
# empty string does.
# ---------------------------------------------------------------------------
RISK_REGISTER_REQUIRED_PRESENT = (
    "risk_id", "rmf_function", "rmf_category", "likelihood", "impact", "severity",
)
RISK_REGISTER_REQUIRED_NONEMPTY = (
    "title", "system", "description", "source_evidence", "current_controls",
    "residual_risk", "owner", "status", "review_date",
)

_MODULE_CITATION = re.compile(r"AIE-304-M0[124]\b")
_FINDING_CITATION = re.compile(r"\bF-04-0\d\b")
_LAB_CITATION = re.compile(r"\bLab M0[12]-[A-D]\b")
_CODE_LOCATOR = re.compile(r"\w+\.py::\w+")
_MEASURED_NUMBER = re.compile(r"\d+(\.\d+)?\s*%|\b\d+/\d+\b")


def check_evidence_traceable(evidence: str) -> bool:
    """True iff `evidence` points at something a reviewer could actually
    re-check:
      - a filed AIE-304-M04 finding ID (self-contained -- the finding
        record itself carries the measured numbers), OR
      - an AIE-304-M01/M02 lab PLUS a stated measured rate (a lab has more
        than one result in it, so which one has to be said), OR
      - a fresh file::function locator PLUS a stated measured rate (new
        evidence -- e.g. from this module's own Assessment -- needs its
        result stated to mean anything, since there is no filed ID to
        point at yet).
    A bare appeal to unrecorded testing, a course-module number with no
    specific finding or lab, or a code pointer with no stated result, all
    fail -- and so does a *fabricated* lab or finding ID that merely has
    the right shape (see Lesson 4)."""
    if _MODULE_CITATION.search(evidence) and _FINDING_CITATION.search(evidence):
        return True
    if (
        _MODULE_CITATION.search(evidence)
        and _LAB_CITATION.search(evidence)
        and _MEASURED_NUMBER.search(evidence)
    ):
        return True
    if _CODE_LOCATOR.search(evidence) and _MEASURED_NUMBER.search(evidence):
        return True
    return False


def validate_risk_entry(entry: dict) -> list[str]:
    """Return the required fields that are missing, empty, structurally
    invalid, or -- for source_evidence specifically -- untraceable. An
    empty list means the entry is complete enough to file."""
    problems = []
    for f in RISK_REGISTER_REQUIRED_PRESENT:
        if f not in entry:
            problems.append(f)
    for f in RISK_REGISTER_REQUIRED_NONEMPTY:
        if not entry.get(f):
            problems.append(f)
    if entry.get("rmf_function") not in RMF_FUNCTIONS:
        problems.append("rmf_function:invalid")
    if entry.get("rmf_category") not in RMF_CATEGORIES:
        problems.append("rmf_category:invalid")
    if entry.get("source_evidence") and not check_evidence_traceable(entry["source_evidence"]):
        problems.append("source_evidence:untraceable")
    return problems


def finding_to_risk_entry(
    finding: dict, rmf_function: str, rmf_category: str, owner: str, review_date: str,
) -> dict:
    """Build a risk register entry from an already-filed, already-validated
    AIE-304-M04 finding. severity/likelihood/impact are carried over
    EXACTLY -- never recomputed by hand -- and source_evidence is built
    automatically from the finding's own finding_id, so a correct call is
    traceable by construction."""
    return {
        "risk_id": f"R-{finding['finding_id']}",
        "rmf_function": rmf_function,
        "rmf_category": rmf_category,
        "likelihood": finding["likelihood"],
        "impact": finding["impact"],
        "severity": finding["severity"],
        "title": finding["title"],
        "system": finding["target"],
        "description": finding["root_cause"],
        "source_evidence": (
            f"AIE-304-M04 finding {finding['finding_id']} "
            f"({finding['reproduction']}; asr={finding['asr']}, "
            f"trials_n={finding['trials_n']})."
        ),
        "current_controls": finding.get("recommended_remediation") or "none filed",
        "residual_risk": finding["residual_risk"],
        "owner": owner,
        "status": finding.get("status", "open"),
        "review_date": review_date,
    }


def measurement_to_risk_entry(
    *, risk_id: str, rmf_function: str, rmf_category: str, title: str, system: str,
    description: str, source_evidence: str, current_controls: str, residual_risk: str,
    owner: str, review_date: str, impact: str, likelihood: str,
) -> dict:
    """Build a risk register entry from a raw measured lab result (not a
    filed M04 finding) -- e.g. an AIE-304-M01 ASR table or an AIE-304-M02
    before/after number. Unlike finding_to_risk_entry, source_evidence is
    NOT built automatically: the caller must write a citation that passes
    check_evidence_traceable() themselves, because there is no finding_id
    to derive it from. severity is computed, never passed in."""
    return {
        "risk_id": risk_id,
        "rmf_function": rmf_function,
        "rmf_category": rmf_category,
        "likelihood": likelihood,
        "impact": impact,
        "severity": rate_severity(impact, likelihood),
        "title": title,
        "system": system,
        "description": description,
        "source_evidence": source_evidence,
        "current_controls": current_controls,
        "residual_risk": residual_risk,
        "owner": owner,
        "status": "open",
        "review_date": review_date,
    }


def audit_full_packet(risk_register: list[dict]) -> dict[str, list[str]]:
    """Run validate_risk_entry over every entry; return only the entries
    with at least one problem, keyed by risk_id (or a positional key if
    risk_id itself is what's missing). An empty dict means every row in
    the register is filed clean."""
    failures: dict[str, list[str]] = {}
    for i, entry in enumerate(risk_register):
        problems = validate_risk_entry(entry)
        if problems:
            key = entry.get("risk_id") or f"<entry {i} missing risk_id>"
            failures[key] = problems
    return failures

def classify_eu_ai_act_tier(system: dict) -> str:
    """system: {"annex_iii_domain": str | None, "prohibited_practice": str | None,
    "interacts_with_natural_persons": bool}. Returns one of the Act's own
    four tiers, checked in the Act's own precedence order: a prohibited
    practice is disqualifying regardless of domain; a high-risk domain
    governs regardless of whether the system also happens to chat with
    people; a system that merely talks to people, in a domain the Act does
    not name, is "limited"; everything else is "minimal"."""
    if system.get("prohibited_practice") in ARTICLE_5_PROHIBITED_PRACTICES:
        return "unacceptable"
    if system.get("annex_iii_domain") in ANNEX_III_HIGH_RISK_DOMAINS:
        return "high-risk"
    if system.get("interacts_with_natural_persons"):
        return "limited"
    return "minimal"


def article_50_disclosure_required(tier: str) -> bool:
    """Transparency toward the end user is not gated on high-risk status --
    it applies whenever a system talks to a natural person at all, and it
    STACKS with high-risk obligations rather than being replaced by them."""
    return tier in ("limited", "high-risk")

ANNEX_IV_SECTIONS = {
    "1": "General description of the AI system: purpose, provider, "
         "versions, intended users, how it is placed on the market.",
    "2": "Detailed description of the system's elements and development "
         "process: design specs, architecture, data sources and "
         "provenance, validation/testing procedures and their results "

    "3": "Detailed information on monitoring, functioning, and control of "

    "4": "Description of the risk-management system, per Article 9.",
    "5": "Description of relevant changes made through the system's "
         "lifecycle.",
    "6": "List of harmonised standards applied, in full or in part.",
    "7": "Copy of the EU declaration of conformity.",
    "8": "Description of the post-market monitoring system, per Article 72, "
         "including the post-market monitoring plan.",
}


def required_annex_iv_sections(tier: str) -> tuple[str, ...]:
    """Full Annex IV technical documentation is a legal requirement only
    for 'high-risk' systems. 'limited'-tier systems carry a narrower,
    SEPARATE legal obligation (Article 50 disclosure) -- sections 1, 2, and
    4 are still recommended here as engineering best practice (this is
    what MAP and MEASURE ask for regardless of legal tier), but 3 and 5-8
    (formal monitoring/control documentation, the harmonised-standards
    list, the declaration of conformity, the Article 72 post-market
    monitoring PLAN) are specific to the high-risk conformity-assessment
    obligation and do not apply outside it. 'unacceptable' is not a
    documentation question -- Article 5 prohibits placing the system on
    the market at all, so no amount of paperwork substitutes."""
    if tier == "high-risk":
        return tuple(ANNEX_IV_SECTIONS.keys())
    if tier == "limited":
        return ("1", "2", "4")
    if tier == "unacceptable":
        return ()
    return ("1",)


@dataclass
class ModelCard:
    system_name: str
    tier: str
    sections: dict = field(default_factory=dict)
    disclosure_text: str = ""


def build_model_card(system: dict, sections_content: dict) -> ModelCard:
    """Classify the system, determine which Annex IV sections its tier
    actually requires, and refuse to build an incomplete card for that
    tier -- a model card missing a legally required section is not a draft,
    it is a gap that will be found in an audit."""
    tier = classify_eu_ai_act_tier(system)
    required = required_annex_iv_sections(tier)
    missing = [s for s in required if not sections_content.get(s)]
    if missing:
        raise ValueError(
            f"model card missing required Annex IV section(s) for tier "
            f"{tier!r}: {missing}"
        )
    card = ModelCard(system_name=system["name"], tier=tier, sections=dict(sections_content))
    if article_50_disclosure_required(tier):
        card.disclosure_text = (
            f"You are interacting with an AI system ({system['name']}). "
            "It can make mistakes; a human is available on request."
        )
    return card
```

## Lessons

### Lesson 1 — The risk register as wiring, not a poster

**Teaching text.** Search "NIST AI RMF" and the first results are a diagram: four ovals — GOVERN, MAP, MEASURE, MANAGE — arranged in a circle with arrows. Treated as a poster, that diagram teaches nothing actionable; treated as a wiring diagram, it says something specific: **GOVERN** is the standing structure (who is accountable, what risk tolerance means here) that has to exist before the other three mean anything; **MAP** is where you write down what the system is for and what could go wrong with it, before you have numbers; **MEASURE** is where "could go wrong" becomes a trial count and a rate — this is where AIE-304-M01's ASR tables and AIE-304-M04's filed findings already live, whether or not anyone called them that; **MANAGE** is where a measured risk gets a decision, an owner, and a residual-risk statement — this is where AIE-304-M02's defenses and their honest gaps already live. The four functions are not phases you walk through once. A single system has entries in all four, continuously, and a risk register's job is to be the place those entries live as structured data instead of scattered across four separate course modules' worth of prose.

**Worked example.** Take one already-measured fact from each of three earlier modules and place it in the RMF, using `governance.py`'s real category IDs:

| Evidence | Course source | RMF function | RMF category |
|---|---|---|---|
| Indirect injection via `doc_4_poisoned` leaks the secret on 5/5 topic-relevant queries, 0/5 unrelated ones | AIE-304-M01 Lab M01-A | **map** | `MAP 5.1` — likelihood/magnitude of the impact assessed |
| `F-04-02`: with `require_tool_gate` off, privilege escalation succeeds 2/2 times it is attempted | AIE-304-M04 Lab M04-C | **measure** | `MEASURE 2.7` — security/resilience evaluated with a measured rate |
| Isolating the retrieved document from the directive scan drops `kb_injection` ASR from 5/5 to 0/5 | AIE-304-M02 Lab M02-A | **manage** | `MANAGE 2.3` — a response to a high-priority risk, developed and documented |

Build the second row as an actual risk register entry with `finding_to_risk_entry()`:

```python
from governance import finding_to_risk_entry, F_04_02

entry = finding_to_risk_entry(
    F_04_02, rmf_function="measure", rmf_category="MEASURE 2.7",
    owner="platform-security@northwind.example", review_date="2026-08-12",
)
print(entry["severity"], entry["source_evidence"])
# critical AIE-304-M04 finding F-04-02 (lab_c_severity.py::measure_gate_regression(); asr=0.4, trials_n=5).
```

Nothing about the severity, the impact, or the likelihood was re-typed by hand — they came from `F_04_02` unchanged, exactly the way `MANAGE 2.3`'s own definition requires a documented response to trace back to what was actually found, not to a fresh guess at how bad it was.

**Misconception 1 — "GOVERN, MAP, MEASURE, MANAGE are steps you do in that order."** The worked example puts a Lab M04 fact in MEASURE and a Lab M02 fact — which happened chronologically *before* M04 in this course — in MANAGE. That ordering is deliberate: the RMF's four functions describe four *kinds* of activity a mature program does continuously and in parallel, not four checkpoints on a timeline. A system can have a MANAGE entry (a shipped fix) for a risk whose MEASURE entry (the rigorous, trial-counted version of the same risk) gets filed later, in a red-team engagement — which is exactly this course's own history.

**Misconception 2 — "The risk register is a summary of the model card, or vice versa."** They answer different questions for different audiences. The risk register answers "what could go wrong, how do we know, and what are we doing about it" for an internal owner with a `review_date`. The model card (Lesson 3) answers "what is this system, and what has been established about it" for an external or downstream reader who was not in the room for any of the labs. A risk register entry can be `status: "open"` and still be accurately summarized in a model card's risk-management section — the model card does not require every risk to be closed, it requires every claim in it to be honest about what's open.

**Misconception 3 — "A risk register entry is basically a to-do item with a severity tag."** A to-do item disappears when it's done. A risk register entry with `status: "mitigated"` still carries its `residual_risk` field forward — `F-04-02`'s proposed fix, for instance, is explicitly noted as sharing a weakness (the same static allow-list) with the control it replaces. Closing the finding does not close the register entry; the entry's job continues after the finding's does.

### Lesson 2 — EU AI Act tiers: the domain decides, not the model's capability

**Teaching text.** A common and costly misreading of the EU AI Act is that its four risk tiers — unacceptable, high-risk, limited, minimal — track how *capable* or *large* a model is. They do not. `classify_eu_ai_act_tier()` never looks at parameter count, provider, or how impressive the system seems; it looks at **what the system is used for**. Annex III lists eight specific domains (biometric identification, critical infrastructure, education/vocational-training access, employment decisions, access to essential public/private services — including creditworthiness and insurance-risk scoring — law enforcement, migration/asylum/border control, and administration of justice) where an AI system making or materially influencing a decision is high-risk *by domain*, regardless of whether the underlying system is a hand-rolled rules engine or a frontier model. A support chatbot fielding warranty questions and a support chatbot approving a $2,000 credit-line increase can be the identical codebase, running the identical model, at the identical company — and sit in two different Act tiers, because the second one has walked into Annex III's "access to essential services" domain and the first has not.

**Worked example.** TicketBot, as it exists after AIE-304-M02/M04, only answers support questions and can send an email or write a pre-approved account field — it never decides whether a customer gets money, credit, a job, or a service:

```python
from governance import classify_eu_ai_act_tier, article_50_disclosure_required, TICKETBOT_SYSTEM

tier = classify_eu_ai_act_tier(TICKETBOT_SYSTEM)
print(tier, article_50_disclosure_required(tier))
# limited True
```

`limited`, not `minimal` — because `interacts_with_natural_persons=True` triggers Article 50's transparency obligation regardless of domain: a customer talking to TicketBot has a legal right to know they are talking to an AI system. Now trace a proposed feature by hand: TicketBot's team wants to let VIP customers get an automatic credit-line increase, up to a limit, with no human review:

```python
VIP_CREDIT_SYSTEM = {
    "name": "TicketBot VIP Credit-Line Auto-Adjust",
    "annex_iii_domain": "essential_services_access",
    "prohibited_practice": None,
    "interacts_with_natural_persons": True,
}
print(classify_eu_ai_act_tier(VIP_CREDIT_SYSTEM))
# high-risk
```

Same company, same underlying planner code path, same model. The only thing that changed is what the tool now decides, and that alone moves the system from `limited` to `high-risk` — which, as Lesson 3 will show, is not a bigger version of the same paperwork; it is a different, larger legal obligation.

**Misconception 1 — "Our system is 'just a chatbot,' so it can't be high-risk."** `interacts_with_natural_persons` alone only ever produces `limited`, never `high-risk` — Lesson 2's worked example shows that correctly. But `annex_iii_domain` is checked independently of the chat interface: the credit-line example is still, underneath the chat UI, a system that decides creditworthiness. The interface a system happens to be wrapped in has no bearing on which Annex III domain its *decision* falls into.

**Misconception 2 — "If it's not high-risk, there's nothing to disclose."** `article_50_disclosure_required("limited")` is `True`. The Act's tiers are not "heavily regulated" vs. "unregulated" — `minimal` is the only tier with no Act-specific obligation modeled here, and even most systems that talk to end users, TicketBot included, land in `limited`, which still carries a real, checkable requirement: tell the user they're talking to an AI.

**Misconception 3 — "A prohibited-practice system just gets extra-strict high-risk paperwork."** Read `classify_eu_ai_act_tier()`'s own branch order: `prohibited_practice` is checked *first*, and if it matches, the function returns `"unacceptable"` before the Annex III domain is even inspected — regardless of how well-documented the system is. `required_annex_iv_sections("unacceptable")` returns an empty tuple, not the full eight sections, because more documentation is not the applicable response to a prohibited practice under Article 5. The two tiers "get more paperwork" and "may not be placed on the market at all" are entirely different consequences, and conflating them is exactly the mistake the function's branch order exists to prevent.

### Lesson 3 — The model card as scoped Annex IV documentation

**Teaching text.** "Model card" is used loosely across the industry to mean anything from a one-paragraph capability blurb to a full regulatory filing. This module treats it as the second: a model card is this course's implementation of EU AI Act Annex IV technical documentation, deliberately *scoped to the system's actual tier* rather than written at full length regardless of whether the law requires it. `required_annex_iv_sections()` is the scoping function — it is the difference between a model card that pads itself with sections nobody asked for and one that is complete for what it actually has to be. Sections 1 (general description), 2 (development process, data, and measured validation results), and 4 (risk-management summary) are recommended at every tier that talks to end users, because those are exactly what NIST's MAP and MEASURE functions ask for anyway, framework aside. Sections 3, 5, 6, 7, and 8 — formal monitoring/control documentation, lifecycle changes, the harmonised-standards list, the declaration of conformity, and the Article 72 post-market monitoring plan — are the parts of Annex IV that are specific to the high-risk conformity-assessment regime, and `build_model_card()` will not accept a card missing any of them once a system is classified `high-risk`.

**Worked example.** Populate TicketBot's `limited`-tier card using measurements this course already has on hand — no new testing, no invented numbers:

```python
from governance import build_model_card, TICKETBOT_SYSTEM

sections = {
```

Section 2 above is the section most worth reading twice: every sentence in it is a citation to a specific, re-runnable number from a specific earlier module — 8/8 to 0/8, an RoE reference, two named finding IDs — and not one sentence is a claim like "TicketBot has been extensively security-tested." The second sentence is what a real vendor's public model card usually contains instead of the first; Lesson 4 makes that gap mechanical to catch.

**Misconception 1 — "A model card for a 'limited' system can skip the risk-management section since Annex IV's risk-management point (4) is really a high-risk thing."** `required_annex_iv_sections("limited")` returns `("1", "2", "4")` — section 4 is in the *recommended* set at `limited` tier, not only the mandatory set at `high-risk`. The reasoning: Article 9's risk-management-system requirement is legally mandatory only for high-risk systems, but NIST's MAP/MEASURE functions ask every system, regardless of legal tier, to have mapped and measured its risks — the model card's section 4 exists here as engineering best practice tracking a different framework's expectation, not only the Act's.

**Misconception 2 — "Once `build_model_card()` runs without raising, the card is done."** `build_model_card()` enforces *completeness* (every required section present) — it does not, and cannot, enforce *honesty* (whether what's written in each section actually traces to a measurement). A card can pass validation with a fabricated section 2 the same way `F_04_01`'s finding could have named a root cause that doesn't match its own code. Lesson 4's traceability check is the layer that catches the second failure; `build_model_card()` only catches the first.

**Misconception 3 — "The model card should describe the system as it will be once the open findings are fixed."** Section 2's worked example describes `F-04-02` as `"critical, open"` — present tense, current state, not a forward-looking claim about a planned fix. A model card dated `2026-08-12` that describes a system as though a fix scheduled for next quarter has already landed is not documenting the system; it is documenting a future system that does not yet exist, under the current one's name.

### Lesson 4 — Traceability is enforced by a regex, not by trust

**Teaching text.** Every governance framework this module touches — NIST's RMF, the Act's Annex IV — ultimately asks the same underlying question of any claim in a risk register or model card: *can someone other than the author check this?* `check_evidence_traceable()` is this module's answer, made mechanical instead of aspirational. It treats three kinds of citation differently, and the difference is itself the lesson: a citation to a **filed finding ID** (`F-04-02`) is self-contained, because the finding record already carries its own trial count and rate — citing the ID is enough, the same way citing a court case number is enough for someone who can look the case up. A citation to a **course lab** (`Lab M02-A`) is *not* self-contained the same way, because a lab produces more than one result across its Predict/Build/Verify arc — `check_evidence_traceable()` requires the lab citation *and* a stated rate together, because "see Lab M01-B" could mean any of four techniques' ASR. A citation to **fresh evidence** — a `file.py::function` locator for a measurement this module's own Assessment produces, with no prior finding ID to point at — needs the same pairing, for the same reason.

**Worked example.** Four evidence strings, predicted and then checked:

```python
from governance import check_evidence_traceable

check_evidence_traceable(
    "AIE-304-M04 finding F-04-02: require_tool_gate is a single point of "
    "failure (asr=0.4, trials_n=5, attempted 2/5, succeeded 2/2). "
    "See lab_c_severity.py::measure_gate_regression()."
)
# True -- module + finding ID present; self-contained.

check_evidence_traceable(
    "We ran internal red-team testing over several days and are "
    "confident this risk is adequately mitigated."
)
# False -- no module citation, no finding, no lab, no code locator, no number.

check_evidence_traceable("See AIE-304-M01 Lab M01-Z for the relevant measurement.")
# False -- "Lab M01-Z" is not a real lab (M01 only has labs A-C); the
# citation has the right SHAPE and fails the check anyway, because the
# regex checks the shape a real lab ID actually takes, not just the
# presence of the word "Lab".

check_evidence_traceable(
    "AIE-304-M02 Lab M02-A measured kb_injection asr 5/5 -> 0/5 after "
    "isolate_untrusted=True."
)
# True -- module + real lab ID + a stated rate (5/5, 0/5).
```

The third example is the one worth sitting with. It is not obviously worse prose than the fourth — both name a module and a lab. The difference is entirely mechanical: `Lab M01-Z` does not match `Lab M0[12]-[A-D]`, because AIE-304-M01 never had a Lab M01-D, let alone a Lab M01-Z. A learner (or, in production, an author under deadline pressure) who fabricates a plausible-sounding lab reference for a claim they didn't actually verify is caught by the same regex that accepts a real one — not because the regex is clever, but because it checks the *actual, finite set of things that exist* rather than a pattern that merely looks official.

**Misconception 1 — "If it cites a real module number, it's traceable enough."** `check_evidence_traceable("AIE-304-M01 shows this is a known risk.")` is `False` — a bare module citation with no finding, no lab, and no code locator names a *reading*, not a measurement. `_MODULE_CITATION` matching is necessary in two of the three passing branches, never sufficient on its own.

**Misconception 2 — "The traceability check is a formality that a well-written paragraph should be able to satisfy without a rigid citation format."** This is the exact shape of gaming the check is built to resist: a fluent, confident paragraph with no checkable pointer in it (the second example above) is indistinguishable, to a time-pressed reviewer skimming prose, from one that's actually backed by a measurement. The check does not evaluate writing quality — it evaluates whether a specific, finite thing (`F-04-02`, `Lab M02-A`, `measure_gate_regression()`) is actually named, precisely because writing quality is not evidence.

**Misconception 3 — "A finding ID citation and a lab citation should be held to the same bar."** They are held to *different* bars on purpose (Teaching text, above): a finding ID is one fixed record; a lab is several results across a Predict/Build/Verify arc. Requiring a stated number alongside the lab citation, but not alongside the finding citation, is not an inconsistency in `check_evidence_traceable()` — it is the function correctly treating a fine-grained citation as more informative than a coarse one and asking coarse citations to narrow themselves down.
